cmake_minimum_required(VERSION 3.22)
project(DragonBallAdvancedAdventureRecomp C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/gbarecomp" CACHE PATH "gbarecomp checkout")
set(RECOMP_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/recomp-ui" CACHE PATH "recomp-ui checkout")
set(GBARECOMP_RUNTIME_UI_ROOT "${RECOMP_UI_ROOT}" CACHE PATH "runtime UI checkout")
if(NOT EXISTS "${GBARECOMP_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Fetch gbarecomp first: missing ${GBARECOMP_ROOT}")
endif()
if(NOT EXISTS "${RECOMP_UI_ROOT}/recomp_ui.cmake")
  message(FATAL_ERROR "Fetch recomp-ui first: missing ${RECOMP_UI_ROOT}")
endif()
set(VARIANT dragon_ball_advanced_adventure)
set(VARIANT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/variants/${VARIANT}")
file(GLOB GENERATED CONFIGURE_DEPENDS "${VARIANT_DIR}/generated/*.cpp" "${VARIANT_DIR}/generated/*.c")
if(NOT GENERATED)
  message(FATAL_ERROR "No generated guest C++ found. Run gba_recompile on your verified private ROM first.")
endif()
if(NOT EXISTS "${VARIANT_DIR}/game.toml")
  message(FATAL_ERROR "No private game.toml. Run scripts/prepare_rom.py on your verified ROM.")
endif()
add_subdirectory("${GBARECOMP_ROOT}" gbarecomp_build EXCLUDE_FROM_ALL)
# Only this Android diagnostic branch enables the live on-screen performance HUD.
# The other game/engine branches retain their ordinary hot paths unchanged.
if(ANDROID)
  target_compile_definitions(gbarecomp_runtime PRIVATE GBARECOMP_PERF_HUD=1)
endif()
if(ANDROID)
  add_library(main SHARED src/main.cpp ${GENERATED})
  set(GAME_TARGET main)
else()
  add_executable(DragonBallAdvancedAdventureRecomp src/main.cpp ${GENERATED})
  set(GAME_TARGET DragonBallAdvancedAdventureRecomp)
endif()
# Keep huge dispatch/symbol-map translation units at -O0 to minimize build cost.
# Guest execution is performance-critical: the first-boot -O0 was intentionally
# conservative, but is too slow for gameplay. Start with -O1 for the shards;
# do not change the guest program, runtime semantics, or linker setup.
file(GLOB GUEST_SHARDS CONFIGURE_DEPENDS "${VARIANT_DIR}/generated/recompiled_*.cpp")
set_source_files_properties(${GENERATED} PROPERTIES COMPILE_OPTIONS "-O0;-g0")
set_source_files_properties(${GUEST_SHARDS} PROPERTIES COMPILE_OPTIONS "-O1;-g0")
target_include_directories(${GAME_TARGET} PRIVATE
  "${GBARECOMP_ROOT}/src/armv4t"
  "${GBARECOMP_ROOT}/src/gba"
  "${GBARECOMP_ROOT}/src/runtime"
  "${GBARECOMP_ROOT}/src/debug"
  "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(${GAME_TARGET} PRIVATE
  "-Wl,--start-group" gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t "-Wl,--end-group")
# Generated by prepare_rom.py. Avoid assumptions about the user's ROM revision.
include("${VARIANT_DIR}/cartridge.cmake")
target_compile_definitions(${GAME_TARGET} PRIVATE
  GBARECOMP_DEFAULT_DEBUG_PORT=19890
  GBARECOMP_WINDOW_TITLE="DragonBallAdvancedAdventureRecomp"
  GBARECOMP_DEFAULT_GAME_CONFIG="variants/${VARIANT}/game.toml"
  GBARECOMP_BUILTIN_NAME="Dragon Ball: Advanced Adventure"
  GBARECOMP_BUILTIN_SHA1="${DBAA_ROM_SHA1}"
  GBARECOMP_BUILTIN_CRC32=${DBAA_ROM_CRC32}
  GBARECOMP_BUILTIN_REGION="${DBAA_REGION}")

include("${RECOMP_UI_ROOT}/recomp_ui.cmake")
add_library(game_launcher_ui STATIC src/game_launcher_boot.cpp)
set_target_properties(game_launcher_ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(game_launcher_ui PRIVATE
  "${GBARECOMP_ROOT}/src/runtime" "${GBARECOMP_ROOT}/src/gba" "${CMAKE_CURRENT_SOURCE_DIR}/src")
recomp_target_launcher_ui(game_launcher_ui CONSOLE gba)
if(NOT TARGET SDL2::SDL2)
  find_package(SDL2 REQUIRED)
endif()
target_link_libraries(game_launcher_ui PUBLIC SDL2::SDL2)
target_link_libraries(${GAME_TARGET} PRIVATE game_launcher_ui)
target_compile_definitions(${GAME_TARGET} PRIVATE GBAGAME_RECOMP_UI)
